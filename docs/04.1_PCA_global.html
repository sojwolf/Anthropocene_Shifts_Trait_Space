library(cowplot)

make_pca_biplot <- function(pca_result,
                            gbif_imputed,
                            pcs = c(1, 2),
                            show_loading_labels = TRUE,
                            arrow_scale = 10,
                            label_offset = 0.3) {
  
  # -----------------------------
  # Extract PCA scores
  # -----------------------------
  
  pca_scores <- as.data.frame(pca_result$x[, pcs])
  names(pca_scores) <- c("PCx", "PCy")
  pca_scores$group <- gbif_imputed$group

  # Actual PC names
  pc_x_name <- paste0("PC", pcs[1])
  pc_y_name <- paste0("PC", pcs[2])
  
  # -----------------------------
  # Extract loadings
  # -----------------------------
  
  loadings <- as.data.frame(pca_result$rotation[, pcs])
  names(loadings) <- c("PCx", "PCy")
  loadings$trait <- rownames(loadings)
  
  # Scale loading arrows
  loadings <- loadings %>%
    mutate(
      PCx = PCx * arrow_scale,
      PCy = PCy * arrow_scale,
      arrow_length = sqrt(PCx^2 + PCy^2),
      unit_x = PCx / arrow_length,
      unit_y = PCy / arrow_length,
      label_x = PCx + label_offset * unit_x,
      label_y = PCy + label_offset * unit_y
    )
  
  # Extract numbers from trait names
  loadings$trait <- stringr::str_extract(loadings$trait, "\\d+")
  
  # -----------------------------
  # Base PCA plot
  # -----------------------------
  
  p <- ggplot(pca_scores, aes(x = PCx, y = PCy)) +
  labs(
    x = pc_x_name,
    y = pc_y_name
  ) +
    
    stat_density_2d(
      geom = "polygon",
      aes(
        alpha = ..level..,
        fill = group,
        group = group
      ),
      bins = 10
    ) +

    scale_alpha_continuous(
    range = c(0.03, 0.4),
    guide = "none"
    ) +
    
    scale_fill_manual(
      values = c(
        "all" = "#fc8d62",
        "native" = "#2ed5c9"
      )
    ) +
    
    theme_minimal() +
    
    theme(
      panel.grid = element_blank(),
      legend.position = "right",
      axis.title = element_text(size = 14),
      axis.text = element_text(size = 14),
      panel.border = element_rect(
        color = "darkgrey",
        fill = NA,
        linewidth = 1
      )
    ) +
    
    # Loading arrows
    geom_segment(
      data = loadings,
      aes(
        x = 0,
        y = 0,
        xend = PCx,
        yend = PCy
      ),
      arrow = arrow(length = unit(0.1, "cm")),
      color = "#045A7FFF",
      linewidth = 0.5
    )
  
  # -----------------------------
  # Optional loading labels
  # -----------------------------
  
  if (show_loading_labels) {
    
    p <- p +
      geom_text(
        data = loadings,
        aes(
          x = label_x,
          y = label_y,
          label = trait
        ),
        color = "#2F86AC",
        size = 4,
        hjust = 0.5,
        vjust = 0.5
      )
  }
  
  # -----------------------------
  # Zero lines
  # -----------------------------
  
  p <- p +
    geom_hline(
      yintercept = 0,
      color = "dimgrey",
      linetype = "dashed"
    ) +
    geom_vline(
      xintercept = 0,
      color = "dimgrey",
      linetype = "dashed"
    )
  
  # -----------------------------
  # Marginal density: PCx
  # -----------------------------
  
  xdens <- axis_canvas(p, axis = "x") +
    
    geom_density(
      data = pca_scores,
      aes(x = PCx, fill = group),
      alpha = 0.3,
      linewidth = 0.5,
      adjust = 1.5,
      color = NA
    ) +
    
    scale_fill_manual(
      values = c(
        "all" = "#fc8d62",
        "native" = "#2ed5c9"
      )
    ) +
    
    geom_vline(
      data = pca_scores %>%
        group_by(group) %>%
        summarise(median_PCx = median(PCx, na.rm = TRUE)),
      aes(
        xintercept = median_PCx,
        color = group
      ),
      linetype = "solid",
      linewidth = 1
    ) +
    
    scale_color_manual(
      values = c(
        "all" = "#fc8d62",
        "native" = "#2ed5c9"
      )
    )
  
  # -----------------------------
  # Marginal density: PCy
  # -----------------------------
  
  ydens <- axis_canvas(p, axis = "y", coord_flip = TRUE) +
    
    geom_density(
      data = pca_scores,
      aes(x = PCy, fill = group),
      alpha = 0.3,
      linewidth = 0.5,
      adjust = 1.5,
      color = NA
    ) +
    
    scale_fill_manual(
      values = c(
        "all" = "#fc8d62",
        "native" = "#2ed5c9"
      )
    ) +
    
    geom_vline(
      data = pca_scores %>%
        group_by(group) %>%
        summarise(median_PCy = median(PCy, na.rm = TRUE)),
      aes(
        xintercept = median_PCy,
        color = group
      ),
      linetype = "solid",
      linewidth = 1
    ) +
    
    scale_color_manual(
      values = c(
        "all" = "#fc8d62",
        "native" = "#2ed5c9"
      )
    ) +
    
    coord_flip()
  
  # -----------------------------
  # Combine plot + marginal densities
  # -----------------------------
  
  final_plot <- p %>%
    insert_xaxis_grob(
      xdens,
      grid::unit(1, "in"),
      position = "top"
    ) %>%
    insert_yaxis_grob(
      ydens,
      grid::unit(1, "in"),
      position = "right"
    ) %>%
    ggdraw()
  
  return(final_plot)
}
